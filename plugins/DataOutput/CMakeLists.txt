# FlowCV DataOutput, using the same pinned asio2 and oscpack implementations.
find_package(Threads REQUIRED)
set(DATA_OUTPUT_ASIO_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/asio2/include")
set(DATA_OUTPUT_ASIO_3RD "${CMAKE_CURRENT_SOURCE_DIR}/third_party/asio2/3rd")
foreach(pair "CsvFile;CSV_File" "UdpSend;UDP_Send" "TcpSend;TCP_Send" "OscSend;OSC_Send" "SerialSend;Serial_Send")
    list(GET pair 0 target)
    list(GET pair 1 source)
    add_library(${target} SHARED ${source}.cpp)
    target_link_libraries(${target} PRIVATE VisionFlow::Sdk Threads::Threads)
    target_include_directories(${target} SYSTEM BEFORE PRIVATE ${DATA_OUTPUT_ASIO_INCLUDE} ${DATA_OUTPUT_ASIO_3RD})
    target_compile_definitions(${target} PRIVATE ASIO_STANDALONE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    if(WIN32)
        target_link_libraries(${target} PRIVATE ws2_32 mswsock)
        target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0601)
    endif()
    if(VISIONFLOW_BUILD_STUDIO)
        target_sources(${target} PRIVATE ${source}_qt.cpp)
        target_compile_definitions(${target} PRIVATE DATA_OUTPUT_WITH_QT)
        target_link_libraries(${target} PRIVATE VisionFlow::QtNodeUi)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /bigobj)
    endif()
    set_target_properties(${target} PROPERTIES PREFIX "" SUFFIX ".fp")
    visionflow_register_plugin(${target})
endforeach()
target_sources(OscSend PRIVATE third_party/osc/OscOutboundPacketStream.cpp third_party/osc/OscTypes.cpp)
target_include_directories(OscSend PRIVATE third_party/osc)
target_sources(SerialSend PRIVATE serial_enumerator.cpp)
if(WIN32)
    target_sources(SerialSend PRIVATE win/List_Serial_Ports_Win.cpp)
    target_include_directories(SerialSend PRIVATE win)
    target_link_libraries(SerialSend PRIVATE setupapi advapi32)
elseif(APPLE)
    target_sources(SerialSend PRIVATE macos/List_Serial_Ports_Mac.cpp macos/serial/src/serial.cc macos/serial/src/impl/unix.cc macos/serial/src/impl/list_ports/list_ports_osx.cc)
    target_include_directories(SerialSend PRIVATE macos macos/serial/include)
    target_link_libraries(SerialSend PRIVATE "-framework IOKit" "-framework Foundation")
else()
    target_sources(SerialSend PRIVATE linux/List_Serial_Ports_Linux.cpp)
    target_include_directories(SerialSend PRIVATE linux)
endif()
